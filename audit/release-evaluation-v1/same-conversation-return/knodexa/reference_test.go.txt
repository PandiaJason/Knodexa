package handlers

import (
	"errors"
	"reflect"
	"testing"
)

type transformFunc func(any) (any, error)

func (f transformFunc) Transform(value any) (any, error) { return f(value) }

type observerFuncs struct {
	started  func(any)
	finished func(any)
}

func (o observerFuncs) Started(value any)   { o.started(value) }
func (o observerFuncs) Finished(result any) { o.finished(result) }

func TestHandle(t *testing.T) {
	testHandler(t, Handle)
}

func testHandler(t *testing.T, handle func(any, Transformer, Observer) (any, error)) {
	t.Helper()
	inputPointer := &struct{ Value string }{" raw "}
	for _, input := range []any{" raw Mixed Case ", inputPointer, nil, 42} {
		for _, failure := range []error{nil, errors.New("original error")} {
			for _, observe := range []bool{false, true} {
				result := &struct{ Value string }{"original result"}
				var events []string
				calls := 0
				service := transformFunc(func(value any) (any, error) {
					calls++
					if value != input {
						t.Fatalf("input changed: got %#v, want %#v", value, input)
					}
					events = append(events, "transform")
					return result, failure
				})
				var observer Observer
				if observe {
					observer = observerFuncs{
						started: func(value any) {
							if value != input {
								t.Fatal("Started received a different input")
							}
							events = append(events, "started")
						},
						finished: func(value any) {
							if value != result {
								t.Fatal("Finished received a different result")
							}
							events = append(events, "finished")
						},
					}
				}
				got, err := handle(input, service, observer)
				if got != result || err != failure {
					t.Fatal("result or error identity changed")
				}
				if calls != 1 {
					t.Fatalf("Transform called %d times", calls)
				}
				want := []string{"transform"}
				if observe {
					want = []string{"started", "transform"}
					if failure == nil {
						want = append(want, "finished")
					}
				}
				if !reflect.DeepEqual(events, want) {
					t.Fatalf("events = %v, want %v", events, want)
				}
			}
		}
	}
}

func TestHandlePropagatesPanic(t *testing.T) {
	testHandlerPropagatesPanic(t, Handle)
}

func testHandlerPropagatesPanic(t *testing.T, handle func(any, Transformer, Observer) (any, error)) {
	t.Helper()
	marker := &struct{}{}
	defer func() {
		if got := recover(); got != marker {
			t.Fatalf("panic = %v, want original marker", got)
		}
	}()
	handle(nil, transformFunc(func(any) (any, error) { panic(marker) }), observerFuncs{
		started:  func(any) {},
		finished: func(any) { t.Fatal("Finished called after panic") },
	})
}
