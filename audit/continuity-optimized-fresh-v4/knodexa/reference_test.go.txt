package handlers

import (
	"errors"
	"reflect"
	"testing"
)

type transformerFunc func(any) (any, error)

func (f transformerFunc) Transform(value any) (any, error) { return f(value) }

type observerFuncs struct {
	started  func(any)
	finished func(any)
}

func (o observerFuncs) Started(value any)   { o.started(value) }
func (o observerFuncs) Finished(result any) { o.finished(result) }

func TestHandlePreservesIdentityAndOrder(t *testing.T) {
	input := &struct{ Text string }{"  Mixed Case  "}
	result := &struct{ Value int }{42}
	for _, failure := range []error{nil, errors.New("original failure")} {
		t.Run(map[bool]string{true: "failure", false: "success"}[failure != nil], func(t *testing.T) {
			var events []string
			calls := 0
			service := transformerFunc(func(value any) (any, error) {
				calls++
				if value != input {
					t.Fatal("input identity changed")
				}
				events = append(events, "transform")
				return result, failure
			})
			observer := observerFuncs{
				started: func(value any) {
					if value != input {
						t.Fatal("started input identity changed")
					}
					events = append(events, "started")
				},
				finished: func(value any) {
					if value != result {
						t.Fatal("finished result identity changed")
					}
					events = append(events, "finished")
				},
			}
			got, err := Handle(input, service, observer)
			if got != result || err != failure || calls != 1 {
				t.Fatalf("got (%v, %v), calls %d", got, err, calls)
			}
			want := []string{"started", "transform"}
			if failure == nil {
				want = append(want, "finished")
			}
			if !reflect.DeepEqual(events, want) {
				t.Fatalf("events %v, want %v", events, want)
			}
		})
	}
}

func TestHandleWithoutObserverAndServiceSubstitution(t *testing.T) {
	for _, input := range []any{nil, "  Mixed Case  ", 12, []int{1, 2}} {
		calls := 0
		got, err := Handle(input, transformerFunc(func(value any) (any, error) {
			calls++
			if !reflect.DeepEqual(value, input) {
				t.Fatal("raw value changed")
			}
			return value, nil
		}), nil)
		if !reflect.DeepEqual(got, input) || err != nil || calls != 1 {
			t.Fatalf("got (%v, %v), calls %d", got, err, calls)
		}
	}
}

func TestHandlePropagatesPanic(t *testing.T) {
	marker := &struct{}{}
	defer func() {
		if got := recover(); got != marker {
			t.Fatalf("panic %v, want original marker", got)
		}
	}()
	Handle(nil, transformerFunc(func(any) (any, error) { panic(marker) }), nil)
}
