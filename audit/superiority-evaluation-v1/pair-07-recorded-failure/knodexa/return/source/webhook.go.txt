package handlers

import "identitysample/contracts"

// HandleWebhook delivers the original value using the established handler contract.
// Service and observer panics propagate to the caller without retrying delivery.
func HandleWebhook(value any, service Transformer, observer Observer) (any, error) {
	return Handle(value, service, observer)
}

// HandlerContractVersion identifies the delivery contract for SDK integrations.
func HandlerContractVersion() string {
	return contracts.Version
}
