package handlers

import (
	"errors"
	"reflect"
	"testing"
)

func TestHandleWebhook(t *testing.T) {
	input := &struct{ N int }{1}
	resultValue := &struct{ N int }{2}
	originalError := errors.New("delivery failed")
	for _, value := range []any{nil, "", " raw ", 17, input} {
		for _, serviceError := range []error{nil, originalError} {
			for _, observed := range []bool{false, true} {
				events := []string{}
				service := &auditService{events: &events, result: resultValue, err: serviceError}
				observer := &auditObserver{events: &events}
				var hook Observer
				if observed {
					hook = observer
				}
				result, err := HandleWebhook(value, service, hook)
				if result != resultValue || err != serviceError || service.calls != 1 || service.received != value {
					t.Fatalf("delegation changed for %v: result=%v err=%v calls=%d input=%v", value, result, err, service.calls, service.received)
				}
				want := []string{"transform"}
				if observed {
					want = []string{"started", "transform"}
					if observer.started != value {
						t.Fatal("Started input changed")
					}
					if serviceError == nil {
						want = append(want, "finished")
						if observer.finished != resultValue {
							t.Fatal("Finished result changed")
						}
					}
				}
				if !reflect.DeepEqual(events, want) {
					t.Fatalf("events=%v want=%v", events, want)
				}
			}
		}
	}
}

type panicObserver struct {
	auditObserver
	stage string
	value any
}

func (o *panicObserver) Started(value any) {
	o.auditObserver.Started(value)
	if o.stage == "started" {
		panic(o.value)
	}
}

func (o *panicObserver) Finished(value any) {
	o.auditObserver.Finished(value)
	if o.stage == "finished" {
		panic(o.value)
	}
}

func TestHandleWebhookPanics(t *testing.T) {
	for _, stage := range []string{"started", "transform", "finished"} {
		t.Run(stage, func(t *testing.T) {
			events := []string{}
			sentinel := &struct{ N int }{3}
			service := &auditService{events: &events}
			observer := &panicObserver{auditObserver: auditObserver{events: &events}, stage: stage, value: sentinel}
			want := []string{"started"}
			calls := 0
			if stage != "started" {
				calls = 1
				want = append(want, "transform")
			}
			if stage == "transform" {
				service.panicValue = sentinel
			}
			if stage == "finished" {
				want = append(want, "finished")
			}
			defer func() {
				if recovered := recover(); recovered != sentinel {
					t.Fatalf("panic identity changed: %v", recovered)
				}
				if service.calls != calls || !reflect.DeepEqual(events, want) {
					t.Fatalf("calls=%d events=%v; want calls=%d events=%v", service.calls, events, calls, want)
				}
			}()
			HandleWebhook(" raw ", service, observer)
		})
	}
}
