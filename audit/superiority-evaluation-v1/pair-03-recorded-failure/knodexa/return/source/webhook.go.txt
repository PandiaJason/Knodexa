package handlers

import "identitysample/contracts"

// HandleWebhook delivers raw input using the established handler contract.
// Observer and service panics propagate unchanged; delivery is never retried.
func HandleWebhook(value any, service Transformer, observer Observer) (any, error) {
	return Handle(value, service, observer)
}

// HandlerContractVersion identifies the production handler compatibility contract.
func HandlerContractVersion() string {
	return contracts.Version
}
