package handlers

import "identitysample/contracts"

// HandleWebhook delivers the original value using the established handler contract.
// Observer and service panics propagate to the caller; delivery is never retried.
func HandleWebhook(value any, service Transformer, observer Observer) (any, error) {
	return Handle(value, service, observer)
}

// HandlerContractVersion reports the production handler integration contract.
func HandlerContractVersion() string { return contracts.Version }
