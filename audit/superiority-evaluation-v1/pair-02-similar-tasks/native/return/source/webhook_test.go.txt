package handlers

import (
	"errors"
	"reflect"
	"testing"
)

func TestWebhookRawDelegation(t *testing.T) {
	input := &struct{ N int }{3}
	for _, value := range []any{"hello", " hello ", "", nil, 17, input} {
		for _, observed := range []bool{false, true} {
			for _, failure := range []error{nil, errors.New("original")} {
				events := []string{}
				resultValue := &struct{ N int }{7}
				service := &auditService{events: &events, result: resultValue, err: failure}
				observer := &auditObserver{events: &events}
				var hook Observer
				if observed {
					hook = observer
				}
				result, err := HandleWebhook(value, service, hook)
				if result != resultValue || err != failure || service.calls != 1 || service.received != value {
					t.Fatal("raw delegation or result/error identity changed")
				}
				want := []string{"transform"}
				if observed {
					want = []string{"started", "transform"}
					if failure == nil {
						want = append(want, "finished")
						if observer.finished != resultValue {
							t.Fatal("finished payload changed")
						}
					}
					if observer.started != value {
						t.Fatal("started payload changed")
					}
				}
				if !reflect.DeepEqual(events, want) {
					t.Fatalf("events = %v, want %v", events, want)
				}
			}
		}
	}
}

type panicObserver struct {
	*auditObserver
	stage string
	value any
}

func (o *panicObserver) Started(value any) {
	o.auditObserver.Started(value)
	if o.stage == "started" {
		panic(o.value)
	}
}

func (o *panicObserver) Finished(value any) {
	o.auditObserver.Finished(value)
	if o.stage == "finished" {
		panic(o.value)
	}
}

func TestWebhookPanics(t *testing.T) {
	for _, stage := range []string{"started", "transform", "finished"} {
		t.Run(stage, func(t *testing.T) {
			events := []string{}
			sentinel := &struct{ N int }{13}
			service := &auditService{events: &events}
			observer := &panicObserver{auditObserver: &auditObserver{events: &events}, stage: stage, value: sentinel}
			want := []string{"started"}
			calls := 0
			if stage != "started" {
				calls = 1
				want = append(want, "transform")
			}
			if stage == "transform" {
				service.panicValue = sentinel
			}
			if stage == "finished" {
				want = append(want, "finished")
			}
			defer func() {
				if recover() != sentinel || service.calls != calls || !reflect.DeepEqual(events, want) {
					t.Fatalf("panic identity, calls or order changed: calls=%d events=%v", service.calls, events)
				}
			}()
			HandleWebhook("raw", service, observer)
		})
	}
}
