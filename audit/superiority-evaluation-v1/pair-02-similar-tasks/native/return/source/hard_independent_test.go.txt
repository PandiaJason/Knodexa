import ("errors";"reflect";"testing")
type hardHarnessService struct {events *[]string; received any; result any; err error; panicValue any; calls int}
func(s *hardHarnessService)Transform(value any)(any,error){s.calls++;s.received=value;*s.events=append(*s.events,"transform");if s.panicValue!=nil {panic(s.panicValue)};return s.result,s.err}
type hardHarnessObserver struct{events *[]string;started,finished any}
func(o *hardHarnessObserver)Started(value any){o.started=value;*o.events=append(*o.events,"started")}
func(o *hardHarnessObserver)Finished(value any){o.finished=value;*o.events=append(*o.events,"finished")}
