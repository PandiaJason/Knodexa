package handlers

import (
	"errors"
	"reflect"
	"testing"
)

func TestWebhookDelegation(t *testing.T) {
	pointer := &struct{ N int }{42}
	for _, value := range []any{"hello", " hello ", "", nil, 17, pointer} {
		for _, observed := range []bool{false, true} {
			for _, serviceErr := range []error{nil, errors.New("original")} {
				events := []string{}
				resultValue := &struct{ N int }{7}
				service := &auditService{events: &events, result: resultValue, err: serviceErr}
				observer := &auditObserver{events: &events}
				var hook Observer
				if observed {
					hook = observer
				}
				result, err := HandleWebhook(value, service, hook)
				if result != resultValue || err != serviceErr || service.calls != 1 || service.received != value {
					t.Fatalf("delegation mismatch for %v, observed=%v, error=%v", value, observed, serviceErr)
				}
				want := []string{"transform"}
				if observed {
					want = []string{"started", "transform"}
					if observer.started != value {
						t.Fatal("Started payload changed")
					}
					if serviceErr == nil {
						want = append(want, "finished")
						if observer.finished != resultValue {
							t.Fatal("Finished payload changed")
						}
					}
				}
				if !reflect.DeepEqual(events, want) {
					t.Fatalf("events = %v, want %v", events, want)
				}
			}
		}
	}
}

type webhookPanicObserver struct {
	auditObserver
	stage string
	value any
}

func (o *webhookPanicObserver) Started(value any) {
	o.auditObserver.Started(value)
	if o.stage == "started" {
		panic(o.value)
	}
}

func (o *webhookPanicObserver) Finished(value any) {
	o.auditObserver.Finished(value)
	if o.stage == "finished" {
		panic(o.value)
	}
}

func TestWebhookPanics(t *testing.T) {
	for _, stage := range []string{"started", "transform", "finished"} {
		t.Run(stage, func(t *testing.T) {
			events := []string{}
			panicValue := &struct{ N int }{13}
			service := &auditService{events: &events, result: "result"}
			observer := &webhookPanicObserver{auditObserver: auditObserver{events: &events}, stage: stage, value: panicValue}
			want := []string{"started"}
			calls := 0
			if stage != "started" {
				calls = 1
				want = append(want, "transform")
			}
			if stage == "transform" {
				service.panicValue = panicValue
			}
			if stage == "finished" {
				want = append(want, "finished")
			}
			defer func() {
				if recovered := recover(); recovered != panicValue {
					t.Fatalf("panic = %v, want original panic", recovered)
				}
				if service.calls != calls || !reflect.DeepEqual(events, want) {
					t.Fatalf("calls = %d, events = %v; want %d, %v", service.calls, events, calls, want)
				}
			}()
			HandleWebhook("input", service, observer)
		})
	}
}
