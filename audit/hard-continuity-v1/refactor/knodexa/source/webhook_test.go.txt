package handlers

import (
	"errors"
	"reflect"
	"testing"
)

func TestHandleWebhook(t *testing.T) {
	for _, observed := range []bool{false, true} {
		for _, value := range []any{"hello", " hello ", "", nil, 17} {
			for _, serviceErr := range []error{nil, errors.New("original")} {
				events := []string{}
				resultValue := &struct{ N int }{7}
				service := &auditService{events: &events, result: resultValue, err: serviceErr}
				observer := &auditObserver{events: &events}
				var hook Observer
				if observed {
					hook = observer
				}
				result, err := HandleWebhook(value, service, hook)
				if result != resultValue || err != serviceErr || service.calls != 1 || !reflect.DeepEqual(service.received, value) {
					t.Fatal("raw delegation or result/error identity changed")
				}
				want := []string{"transform"}
				if observed {
					want = []string{"started", "transform"}
					if !reflect.DeepEqual(observer.started, value) {
						t.Fatal("Started payload changed")
					}
					if serviceErr == nil {
						want = append(want, "finished")
						if observer.finished != resultValue {
							t.Fatal("Finished payload changed")
						}
					}
				}
				if !reflect.DeepEqual(events, want) {
					t.Fatalf("events: %v, want %v", events, want)
				}
			}
		}
	}
}

type panickingObserver struct {
	stage  string
	value  any
	events *[]string
}

func (o *panickingObserver) Started(any) {
	*o.events = append(*o.events, "started")
	if o.stage == "started" {
		panic(o.value)
	}
}
func (o *panickingObserver) Finished(any) {
	*o.events = append(*o.events, "finished")
	if o.stage == "finished" {
		panic(o.value)
	}
}

func TestHandleWebhookPanics(t *testing.T) {
	for _, stage := range []string{"started", "transform", "finished"} {
		t.Run(stage, func(t *testing.T) {
			events := []string{}
			panicValue := &struct{ N int }{13}
			service := &auditService{events: &events}
			observer := &panickingObserver{stage: stage, value: panicValue, events: &events}
			want := []string{"started"}
			calls := 0
			if stage != "started" {
				want = append(want, "transform")
				calls = 1
			}
			if stage == "transform" {
				service.panicValue = panicValue
			}
			if stage == "finished" {
				want = append(want, "finished")
			}
			defer func() {
				if recover() != panicValue {
					t.Fatal("panic identity changed")
				}
				if service.calls != calls || !reflect.DeepEqual(events, want) {
					t.Fatalf("calls=%d events=%v", service.calls, events)
				}
			}()
			HandleWebhook("x", service, observer)
		})
	}
}
