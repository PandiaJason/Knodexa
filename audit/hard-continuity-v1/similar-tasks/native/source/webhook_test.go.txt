package handlers

import (
	"errors"
	"reflect"
	"testing"
)

func TestWebhookDelegation(t *testing.T) {
	if got := HandlerContractVersion(); got != "raw-v1" {
		t.Fatalf("contract version = %q", got)
	}
	inputPointer := &struct{ N int }{1}
	originalError := errors.New("service error")
	for _, value := range []any{"hello", " hello ", "", nil, 17, inputPointer} {
		for _, observed := range []bool{false, true} {
			for _, serviceError := range []error{nil, originalError} {
				events := []string{}
				resultPointer := &struct{ N int }{2}
				service := &auditService{events: &events, result: resultPointer, err: serviceError}
				observer := &auditObserver{events: &events}
				var hook Observer
				if observed {
					hook = observer
				}
				result, err := HandleWebhook(value, service, hook)
				if result != resultPointer || err != serviceError || service.calls != 1 || service.received != value {
					t.Fatalf("delegation mismatch for %v", value)
				}
				want := []string{"transform"}
				if observed {
					want = []string{"started", "transform"}
					if observer.started != value {
						t.Fatal("Started input changed")
					}
					if serviceError == nil {
						want = append(want, "finished")
						if observer.finished != resultPointer {
							t.Fatal("Finished result changed")
						}
					}
				}
				if !reflect.DeepEqual(events, want) {
					t.Fatalf("events = %v, want %v", events, want)
				}
			}
		}
	}
}

type panicObserver struct {
	auditObserver
	stage string
	value any
}

func (o *panicObserver) Started(value any) {
	o.auditObserver.Started(value)
	if o.stage == "started" {
		panic(o.value)
	}
}

func (o *panicObserver) Finished(value any) {
	o.auditObserver.Finished(value)
	if o.stage == "finished" {
		panic(o.value)
	}
}

func TestWebhookPanics(t *testing.T) {
	for _, stage := range []string{"started", "transform", "finished"} {
		t.Run(stage, func(t *testing.T) {
			events := []string{}
			panicValue := &struct{ N int }{3}
			service := &auditService{events: &events, result: "result"}
			observer := &panicObserver{auditObserver: auditObserver{events: &events}, stage: stage, value: panicValue}
			wantCalls := 1
			wantEvents := []string{"started", "transform"}
			if stage == "started" {
				wantCalls = 0
				wantEvents = []string{"started"}
			} else if stage == "transform" {
				service.panicValue = panicValue
			} else {
				wantEvents = append(wantEvents, "finished")
			}
			defer func() {
				if got := recover(); got != panicValue {
					t.Fatalf("panic = %v, want original %v", got, panicValue)
				}
				if service.calls != wantCalls || !reflect.DeepEqual(events, wantEvents) {
					t.Fatalf("calls = %d, events = %v", service.calls, events)
				}
			}()
			HandleWebhook("input", service, observer)
		})
	}
}
