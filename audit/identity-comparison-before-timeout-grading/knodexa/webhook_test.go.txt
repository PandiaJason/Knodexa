package handlers

import (
	"errors"
	"reflect"
	"testing"
)

func TestHandleWebhookPreservesIdentityAndOrdering(t *testing.T) {
	input := &struct{ Delivery string }{" raw delivery "}
	result := &struct{ Status string }{" original result "}
	failure := errors.New("delivery failure")
	for _, tc := range []struct {
		name string
		err  error
	}{{"success", nil}, {"failure", failure}} {
		t.Run(tc.name, func(t *testing.T) {
			var events []string
			calls := 0
			service := transformerFunc(func(value any) (any, error) {
				calls++
				if value != input {
					t.Fatal("input identity changed")
				}
				events = append(events, "transform")
				return result, tc.err
			})
			observer := observerFuncs{
				started: func(value any) {
					if value != input {
						t.Fatal("Started input identity changed")
					}
					events = append(events, "started")
				},
				finished: func(value any) {
					if value != result {
						t.Fatal("Finished result identity changed")
					}
					events = append(events, "finished")
				},
			}
			got, err := HandleWebhook(input, service, observer)
			if got != result || err != tc.err || calls != 1 {
				t.Fatalf("got (%v, %v), calls %d", got, err, calls)
			}
			want := []string{"started", "transform"}
			if tc.err == nil {
				want = append(want, "finished")
			}
			if !reflect.DeepEqual(events, want) {
				t.Fatalf("events %v, want %v", events, want)
			}
		})
	}
}

func TestHandleWebhookRawValuesAndServiceSubstitution(t *testing.T) {
	for _, input := range []any{nil, " mixed Case\t delivery ", 42, []byte(" raw "), map[string]any{"delivery": " raw "}} {
		calls := 0
		got, err := HandleWebhook(input, transformerFunc(func(value any) (any, error) {
			calls++
			if !reflect.DeepEqual(value, input) {
				t.Fatal("raw input changed")
			}
			return value, nil
		}), nil)
		if !reflect.DeepEqual(got, input) || err != nil || calls != 1 {
			t.Fatalf("got (%v, %v), calls %d", got, err, calls)
		}
	}
	got, err := HandleWebhook(nil, transformerFunc(func(any) (any, error) { return "alternate", nil }), nil)
	if got != "alternate" || err != nil {
		t.Fatalf("substitute got (%v, %v)", got, err)
	}
}

func TestHandleWebhookPropagatesPanic(t *testing.T) {
	sentinel := &struct{ Message string }{"delivery panic"}
	var events []string
	defer func() {
		if got := recover(); got != sentinel {
			t.Fatalf("panic %v, want %v", got, sentinel)
		}
		if !reflect.DeepEqual(events, []string{"started", "transform"}) {
			t.Fatalf("events %v", events)
		}
	}()
	HandleWebhook(nil, transformerFunc(func(any) (any, error) {
		events = append(events, "transform")
		panic(sentinel)
	}), observerFuncs{
		started:  func(any) { events = append(events, "started") },
		finished: func(any) { t.Fatal("Finished called after panic") },
	})
}
