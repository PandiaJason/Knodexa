package handlers

import (
	"errors"
	"reflect"
	"testing"
)

func TestHandleWebhook(t *testing.T) {
	inputPointer := &struct{ delivery string }{" raw "}
	resultPointer := &struct{ delivery string }{" result "}
	sentinel := errors.New("delivery error")
	for _, input := range []any{" raw Mixed Case ", inputPointer, 42, nil} {
		for _, output := range []any{resultPointer, nil} {
			for _, serviceError := range []error{nil, sentinel} {
				for _, observed := range []bool{false, true} {
					calls := 0
					var events []string
					var observer Observer
					if observed {
						observer = observerFuncs{
							started: func(value any) {
								if value != input {
									t.Fatalf("Started input changed: %#v", value)
								}
								events = append(events, "started")
							},
							finished: func(value any) {
								if value != output {
									t.Fatal("Finished result changed")
								}
								events = append(events, "finished")
							},
						}
					}
					result, err := HandleWebhook(input, transformFunc(func(value any) (any, error) {
						calls++
						if value != input {
							t.Fatalf("Transform input changed: %#v", value)
						}
						events = append(events, "transform")
						return output, serviceError
					}), observer)
					if calls != 1 || result != output || err != serviceError {
						t.Fatalf("calls=%d result=%v error=%v", calls, result, err)
					}
					want := []string{"transform"}
					if observed {
						want = []string{"started", "transform"}
						if serviceError == nil {
							want = append(want, "finished")
						}
					}
					if !reflect.DeepEqual(events, want) {
						t.Fatalf("events=%v want=%v", events, want)
					}
				}
			}
		}
	}
}

func TestHandleWebhookPropagatesPanic(t *testing.T) {
	panicValue := &struct{}{}
	calls := 0
	defer func() {
		if recovered := recover(); recovered != panicValue {
			t.Fatalf("panic changed or recovered: %v", recovered)
		}
		if calls != 1 {
			t.Fatalf("Transform called %d times", calls)
		}
	}()
	HandleWebhook(" raw ", transformFunc(func(any) (any, error) {
		calls++
		panic(panicValue)
	}), observerFuncs{
		started:  func(any) {},
		finished: func(any) { t.Fatal("Finished called after panic") },
	})
}
