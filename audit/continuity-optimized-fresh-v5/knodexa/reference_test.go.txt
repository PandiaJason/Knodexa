package handlers

import (
	"errors"
	"reflect"
	"testing"
)

type transformFunc func(any) (any, error)

func (f transformFunc) Transform(value any) (any, error) { return f(value) }

type observerFuncs struct {
	started  func(any)
	finished func(any)
}

func (o observerFuncs) Started(value any)   { o.started(value) }
func (o observerFuncs) Finished(result any) { o.finished(result) }

func TestHandleOrderingAndIdentity(t *testing.T) {
	input := &struct{ Value string }{" raw input "}
	result := &struct{ Value string }{" raw result "}
	for _, serviceErr := range []error{nil, errors.New("original error")} {
		var events []string
		calls := 0
		service := transformFunc(func(value any) (any, error) {
			calls++
			events = append(events, "transform")
			if value != input {
				t.Fatal("input identity changed")
			}
			return result, serviceErr
		})
		observer := observerFuncs{
			started: func(value any) {
				events = append(events, "started")
				if value != input {
					t.Fatal("Started input identity changed")
				}
			},
			finished: func(value any) {
				events = append(events, "finished")
				if value != result {
					t.Fatal("Finished result identity changed")
				}
			},
		}
		got, err := Handle(input, service, observer)
		if got != result || err != serviceErr {
			t.Fatal("result or error identity changed")
		}
		if calls != 1 {
			t.Fatalf("Transform called %d times", calls)
		}
		want := []string{"started", "transform"}
		if serviceErr == nil {
			want = append(want, "finished")
		}
		if !reflect.DeepEqual(events, want) {
			t.Fatalf("events = %v, want %v", events, want)
		}
	}
}

func TestHandleWithoutObserver(t *testing.T) {
	for _, input := range []any{nil, " mixed Case ", 42} {
		calls := 0
		got, err := Handle(input, transformFunc(func(value any) (any, error) {
			calls++
			if value != input {
				t.Fatal("raw value changed")
			}
			return value, nil
		}), nil)
		if got != input || err != nil || calls != 1 {
			t.Fatalf("got (%v, %v), calls %d", got, err, calls)
		}
	}
}

func TestHandlePropagatesPanic(t *testing.T) {
	panicValue := &struct{}{}
	var events []string
	defer func() {
		if recover() != panicValue {
			t.Fatal("panic identity changed or panic was recovered")
		}
		if !reflect.DeepEqual(events, []string{"started", "transform"}) {
			t.Fatalf("events = %v", events)
		}
	}()
	Handle(nil, transformFunc(func(any) (any, error) {
		events = append(events, "transform")
		panic(panicValue)
	}), observerFuncs{
		started:  func(any) { events = append(events, "started") },
		finished: func(any) { events = append(events, "finished") },
	})
}
